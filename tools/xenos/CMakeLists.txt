cmake_minimum_required(VERSION 3.25)
project(B3XenosRecomp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REXSDK_DIR "${CMAKE_CURRENT_LIST_DIR}/../../sdk" CACHE PATH "Prepared ReXGlue SDK")
set(DXC_ROOT "" CACHE PATH "Extracted DirectXShaderCompiler release")

find_path(DXC_HEADERS dxcapi.h HINTS "${DXC_ROOT}/inc" "${DXC_ROOT}/include/dxc" REQUIRED)
if(WIN32)
    find_file(DXC_LIBRARY dxcompiler.lib HINTS "${DXC_ROOT}/lib/x64" REQUIRED)
    find_file(DXC_DLL dxcompiler.dll HINTS "${DXC_ROOT}/bin/x64" REQUIRED)
else()
    find_library(DXC_LIBRARY dxcompiler HINTS "${DXC_ROOT}/lib" REQUIRED)
endif()

include(FetchContent)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zstd GIT_REPOSITORY https://github.com/facebook/zstd.git
    GIT_TAG v1.5.7 SOURCE_SUBDIR build/cmake)
FetchContent_MakeAvailable(zstd)
add_subdirectory("${REXSDK_DIR}/thirdparty/fmt" fmt EXCLUDE_FROM_ALL)
add_subdirectory("${REXSDK_DIR}/thirdparty/xxHash/cmake_unofficial" xxhash EXCLUDE_FROM_ALL)

add_executable(XenosRecomp main.cpp dxc_compiler.cpp shader_recompiler.cpp thirdparty/smol-v/source/smolv.cpp)
target_include_directories(XenosRecomp PRIVATE "${DXC_HEADERS}" thirdparty/smol-v/source)
target_compile_definitions(XenosRecomp PRIVATE RENUT_RECOMP)
target_compile_options(XenosRecomp PRIVATE -Wno-switch -Wno-null-arithmetic -fms-extensions)
target_precompile_headers(XenosRecomp PRIVATE pch.h)
target_link_libraries(XenosRecomp PRIVATE "${DXC_LIBRARY}" xxHash::xxhash libzstd_static fmt::fmt)
if(MINGW)
    target_compile_options(XenosRecomp PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/mingw_dxc_uuid.h" -fexperimental-library)
endif()
if(WIN32)
    add_custom_command(TARGET XenosRecomp POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${DXC_DLL}" "$<TARGET_FILE_DIR:XenosRecomp>" VERBATIM)
endif()
